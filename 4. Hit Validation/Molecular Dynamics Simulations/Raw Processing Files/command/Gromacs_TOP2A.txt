# PHẦN 3: ĐÓNG HỘP, SOLVATE VÀ IONIZATION
################################################################################

gmx editconf -f complex.gro -o box.gro -c -d 1.0 -bt triclinic

gmx solvate -cp box.gro -cs spc216.gro -o box_sol.gro -p topol.top

gmx grompp -f ions.mdp -c box_sol.gro -p topol.top -o ions.tpr -maxwarn 1

gmx genion -s ions.tpr -o box_sol_ion.gro -p topol.top -pname NA -nname CL -neutral -conc 0.15
>SOL
